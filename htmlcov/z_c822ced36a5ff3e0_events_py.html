55def _valida_cpf(cpf: str) -> bool: 

56 """Validação completa de CPF: 11 dígitos + dígitos verificadores (H7). 

57 

58 Algoritmo oficial: pesos 10..2 (1º dígito) e 11..2 (2º dígito); 

59 resto < 2 → 0; senão 11 - resto. 

60 """ 

61 cpf_clean = re.sub(r"\D", "", cpf) 

62 if len(cpf_clean) != 11: 

63 return False 

64 if cpf_clean == cpf_clean[0] * 11: 

65 return False 

66 sum1 = sum(int(cpf_clean[i]) * (10 - i) for i in range(9)) 

67 resto1 = sum1 % 11 

68 d1 = 0 if resto1 < 2 else 11 - resto1 

69 if int(cpf_clean[9]) != d1: 

70 return False 

71 sum2 = sum(int(cpf_clean[i]) * (11 - i) for i in range(10)) 

72 resto2 = sum2 % 11 

73 d2 = 0 if resto2 < 2 else 11 - resto2 

74 return int(cpf_clean[10]) == d2

77def _valida_cnpj(cnpj: str) -> bool: 

78 """Validação completa de CNPJ: 14 dígitos + dígitos verificadores (H7).""" 

79 cnpj_clean = re.sub(r"\D", "", cnpj) 

80 if len(cnpj_clean) != 14: 

81 return False 

82 if cnpj_clean == cnpj_clean[0] * 14: 

83 return False 

84 weights1 = [5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2] 

85 weights2 = [6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2] 

86 sum1 = sum(int(cnpj_clean[i]) * weights1[i] for i in range(12)) 

87 d1 = 0 if sum1 % 11 < 2 else 11 - (sum1 % 11) 

88 if int(cnpj_clean[12]) != d1: 

89 return False 

90 sum2 = sum(int(cnpj_clean[i]) * weights2[i] for i in range(13)) 

91 d2 = 0 if sum2 % 11 < 2 else 11 - (sum2 % 11) 

92 return int(cnpj_clean[13]) == d2

95def _valida_data(data: str) -> bool: 

96 """Valida formato DD/MM/AAAA.""" 

97 return bool(re.match(r"^\d{2}/\d{2}/\d{4}$", data))

100def _valida_uf(uf: str) -> bool: 

101 """Valida UF brasileira (2 letras maiúsculas).""" 

102 ufs = { 

103 "AC", "AL", "AP", "AM", "BA", "CE", "DF", "ES", "GO", "MA", "MT", "MS", 

104 "MG", "PA", "PB", "PR", "PE", "PI", "RJ", "RN", "RS", "RO", "RR", "SC", 

105 "SP", "SE", "TO", 

106 } 

107 return uf.upper() in ufs

110def _valida_placa(placa: str) -> bool: 

111 """Valida placa brasileira (Mercosul: ABC1D23 ou antiga: ABC-1234).""" 

112 placa_clean = re.sub(r"[-\s]", "", placa.upper()) 

113 # Mercosul: 3 letras + 1 digito + 3 alfanuméricos 

114 mercosul = bool(re.match(r"^[A-Z]{3}[0-9][A-Z0-9]{3}$", placa_clean)) 

115 # Antiga: 3 letras + 4 digitos 

116 antiga = bool(re.match(r"^[A-Z]{3}[0-9]{4}$", placa_clean)) 

117 return mercosul or antiga

120def _valida_renavam(renavan: str) -> bool: 

121 """Validação básica de RENAVAN (11 dígitos).""" 

122 ren_clean = re.sub(r"\D", "", renavan) 

123 return len(ren_clean) == 11

126def _valida_chassis(chassis: str) -> bool: 

127 """Validação básica de chassi (17 caracteres alfanuméricos, excluindo I, O, Q).""" 

128 ch = re.sub(r"[-\s]", "", chassis.upper()) 

129 if len(ch) != 17: 

130 return False 

131 # VIN não pode ter I, O, Q 

132 invalidos = set("IOQ") 

133 return all(c not in invalidos for c in ch)
