65def _norm(s: str) -> str: 

66 """Normaliza string: sem acentos, sem espacos extras, maiuscula.""" 

67 if s is None: 

68 return "" 

69 if not isinstance(s, str): 

70 s = str(s) 

71 s = unicodedata.normalize("NFKD", s) 

72 s = "".join(c for c in s if not unicodedata.combining(c)) 

73 return " ".join(s.split()).upper()

76def _load_municipios() -> list[dict]: 

77 try: 

78 with open(_MUNICIPIOS_PATH, encoding="utf-8") as f: 

79 data = json.load(f) 

80 except (FileNotFoundError, json.JSONDecodeError): 

81 return [] 

82 return [{"id": int(m["id"]), "nome": m["nome"], "uf": m["uf"], 

83 "capital": int(m.get("capital", 0))} for m in data]

94def validar_uf(uf: Optional[str]) -> bool: 

95 """True se UF for uma das 27 siglas oficiais (case-insensitive).""" 

96 return uf is not None and str(uf).strip().upper() in UFS_SIGLAS

99def municipios_da_uf(uf: Optional[str]) -> list[dict]: 

100 """Lista de municipios de uma UF (id, nome, capital), em ordem alfabetica.""" 

101 if not validar_uf(uf): 

102 return [] 

103 lista = [dict(m) for m in MUNICIPIOS if m["uf"] == uf.strip().upper()] 

104 lista.sort(key=lambda m: (not m["capital"], _norm(m["nome"]))) 

105 return lista

108def validar_cidade(uf: Optional[str], cidade: Optional[str]) -> bool: 

109 """ 

110 Validacao estrita: a cidade precisa existir na tabela oficial da UF. 

111 Case/accent-insensitive. Lista oficial: 5.571 municipios IBGE. 

112 """ 

113 if not validar_uf(uf) or not cidade or not str(cidade).strip(): 

114 return False 

115 return (uf.strip().upper(), _norm(str(cidade))) in _MUNICIPIO_INDEX

117 

118def capital_da_uf(uf: Optional[str]) -> str: 

119 """Nome da capital de uma UF (sem acento).""" 

120 if not validar_uf(uf): 

121 return "" 

122 return CAPITAIS.get(uf.strip().upper(), "")

125def total_municipios() -> int:
