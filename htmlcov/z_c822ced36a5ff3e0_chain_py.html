28class VehicleChain: 

29 """ 

30 Cadeia de blocos para um veículo. 

31 

32 A cadeia começa com um bloco gênesis que representa a fabricação 

33 do veículo (placa, RENAVAN, chassi, especificações técnicas).

59 def create_genesis(self, veiculo_data: dict[str, Any]) -> Block: 

60 """ 

61 Cria o bloco gênesis com dados de fabricação do veículo. 

62 O bloco é obrigatoriamente assinado com ECDSA. 

63 

64 Raises: 

65 ValueError: Se nenhum signer estiver configurado. 

66 

67 Args: 

68 veiculo_data: Dicionário com dados do veículo (placa, RENAVAN, chassi, etc). 

69 

70 Returns: 

71 O bloco gênesis criado. 

72 """ 

73 with self._lock: 

74 return self._create_genesis_impl(veiculo_data) 

75 

76 def _create_genesis_impl(self, veiculo_data: dict[str, Any]) -> Block: 

77 if not self._signer: 

78 raise ValueError("Cadeia requer assinador configurado (set_signer) para registrar operações.") 

79 

80 genesis = Block( 

81 index=0, 

82 timestamp=time.time(), 

83 data={ 

84 "evento_tipo": VehicleEventType.FABRICACAO.value, 

85 "payload": veiculo_data, 

86 "hash_cadeia": "GENESIS", 

87 }, 

88 previous_hash="0" * 64, 

89 difficulty=self.difficulty, 

90 ) 

91 genesis.mine_block() 

92 

93 # Assina obrigatoriamente 

94 sig = self._signer.sign_block(genesis) 

95 genesis.set_signature( 

96 signer_label=sig.signer_label, 

97 signer_pubkey=sig.signer_pubkey, 

98 signature_b64=sig.signature_b64, 

99 signed_at=sig.signed_at, 

100 ) 

101 

102 self.chain.append(genesis) 

103 self._index_event(VehicleEventType.FABRICACAO.value, 0) 

104 return genesis

108 def add_event(self, event_type: str, payload: dict[str, Any]) -> Block: 

109 """ 

110 Adiciona um novo evento à cadeia. 

111 

112 Args: 

113 event_type: Tipo do evento (ex: "COMPRA_VENDA", "SINISTRO"). 

114 payload: Dados do evento. 

115

117 O bloco adicionado. 

118 

119 Raises:

125 def _add_event_impl(self, event_type: str, payload: dict[str, Any]) -> Block: 

126 if not self.chain: 

127 raise ValueError("Cadeia vazia — crie o bloco gênesis primeiro.") 

128 

129 if not self._signer: 

130 raise ValueError("Cadeia requer assinador configurado (set_signer) para registrar operações.") 

131 

132 # Verifica se o evento pode ser adicionado 

133 estado = self.get_estado_atual() 

134 if not VehicleChainProtector.pode_adicionar(event_type, estado.get("situacao", "REGULAR")): 

135 raise ValueError( 

136 f"Evento '{event_type}' bloqueado para veículo com estado '{estado.get('situacao')}'." 

137 ) 

138 

139 last_block = self.chain[-1] 

140 new_block = Block( 

141 index=last_block.index + 1, 

142 timestamp=time.time(), 

143 data={ 

144 "evento_tipo": event_type, 

145 "payload": payload, 

146 "hash_cadeia": last_block.hash, 

147 }, 

148 previous_hash=last_block.hash, 

149 difficulty=self.difficulty, 

150 ) 

151 new_block.mine_block() 

152 

153 # Assina obrigatoriamente 

154 sig = self._signer.sign_block(new_block) 

155 new_block.set_signature( 

156 signer_label=sig.signer_label, 

157 signer_pubkey=sig.signer_pubkey, 

158 signature_b64=sig.signature_b64, 

159 signed_at=sig.signed_at, 

160 ) 

161 

162 self.chain.append(new_block) 

163 self._index_event(event_type, new_block.index) 

164 return new_block

562 def validate(self, require_signatures: bool = True) -> tuple[bool, str]: 

563 """ 

564 Valida toda a cadeia. 

565 

566 Args: 

567 require_signatures: Se True, exige assinatura ECDSA em todos os blocos. 

568 

569 Returns: 

570 Tuple (é_válida, mensagem). 

571 """ 

572 if not self.chain: 

573 return False, "Cadeia vazia." 

574 

575 genesis = self.chain[0] 

576 if genesis.previous_hash != "0" * 64: 

577 return False, "Hash do gênesis inválido." 

578 

579 if not genesis.is_valid(): 

580 return False, f"Bloco gênesis (#{genesis.index}) corrompido." 

581 

582 if require_signatures and not genesis.has_signature(): 

583 return False, f"Bloco gênesis (#{genesis.index}) sem assinatura." 

584 

585 for i in range(1, len(self.chain)): 

586 current = self.chain[i] 

587 previous = self.chain[i - 1] 

588 

589 if current.previous_hash != previous.hash: 

590 return ( 

591 False, 

592 f"Bloco #{current.index}: previous_hash não confere.", 

593 ) 

594 

595 if current.hash != current.compute_hash(): 

596 return ( 

597 False, 

598 f"Bloco #{current.index}: hash próprio não confere.", 

599 ) 

600 

601 if not current.is_valid(): 

602 return ( 

603 False, 

604 f"Bloco #{current.index}: proof-of-work inválido.", 

605 ) 

606 

607 if current.index != i: 

608 return ( 

609 False, 

610 f"Bloco #{current.index}: índice fora de sequência (esperado {i}).", 

611 ) 

612 

613 if require_signatures: 

614 if not current.has_signature(): 

615 return ( 

616 False, 

617 f"Bloco #{current.index}: sem assinatura ECDSA.", 

618 ) 

619 sig = BlockSignature.from_dict(current.signature) 

620 ok, msg = self._verify_signature(current, sig) 

621 if not ok: 

622 return ( 

623 False, 

624 f"Bloco #{current.index}: assinatura inválida — {msg}", 

625 ) 

626 

627 sig_count = sum(1 for b in self.chain if b.has_signature()) 

628 suffix = f" | {sig_count}/{len(self.chain)} assinado(s)" if sig_count else "" 

629 return True, f"Cadeia válida — {len(self.chain)} bloco(s).{suffix}" 

630 

631 def verify_all_signatures(self) -> tuple[bool, str]: 

632 """Verifica todas as assinaturas da cadeia.""" 

633 unsigned = [] 

634 invalid = [] 

635 

636 for block in self.chain: 

637 if not block.has_signature(): 

638 unsigned.append(block.index) 

639 continue 

640 

641 sig = BlockSignature.from_dict(block.signature) 

642 ok, msg = self._verify_signature(block, sig) 

643 if not ok: 

644 invalid.append((block.index, msg)) 

645 

646 lines = [] 

647 if unsigned: 

648 lines.append(f"Sem assinatura: blocos {unsigned}") 

649 if invalid: 

650 for idx, msg in invalid: 

651 lines.append(f"Bloco #{idx}: {msg}") 

652 if not unsigned and not invalid: 

653 lines.append(f"Todas as {len(self.chain)} assinaturas ECDSA são válidas.") 

654 

655 all_valid = not unsigned and not invalid 

656 return all_valid, " | ".join(lines) 

657 

658 def _verify_signature(self, block: Block, sig: BlockSignature) -> tuple[bool, str]: 

659 """Verifica a assinatura de um bloco individual.""" 

660 try: 

661 pubkey = SignatureVerifier._reconstruct_pubkey(sig.signer_pubkey) 

662 return SignatureVerifier.verify_block_signature(block, sig, pubkey) 

663 except Exception as e: 

664 return False, f"Erro na verificação: {e}"

669 """Serializa a cadeia para JSON e salva em disco.""" 

670 data = {

672 "signer_public_key": self._signer.keypair.public_key_pem() if self._signer else None, 

673 "chain": [b.to_dict() for b in self.chain], 

674 } 

675 os.makedirs(os.path.dirname(filepath) or ".", exist_ok=True) 

676 with open(filepath, "w", encoding="utf-8") as f: 

677 json.dump(data, f, indent=2, default=str)

679 @classmethod 

680 def load_from_file(cls, filepath: str) -> "VehicleChain": 

681 """Carrega uma cadeia de um arquivo JSON (restaurando o signer).""" 

682 with open(filepath, "r", encoding="utf-8") as f: 

683 data = json.load(f) 

684 chain = cls(difficulty=data["difficulty"]) 

685 pub_pem = data.get("signer_public_key") 

686 if pub_pem: 

687 try: 

688 chain._signer = Signer(KeyPair.from_public_key_pem(pub_pem)) 

689 except (TypeError, ValueError) as e: 

690 warnings.warn(f"Falha ao restaurar signer publico: {e} — usando signer novo.") 

691 else: 

692 warnings.warn( 

693 "Arquivo sem signer_public_key (formato legado) — " 

694 "gerado novo signer; assinaturas antigas continuam validas." 

695 ) 

696 chain.chain = [Block.from_dict(b) for b in data["chain"]] 

697 for i, block in enumerate(chain.chain): 

698 evento = block.data.get("evento_tipo", "DESCONHECIDO") 

699 chain._index_event(evento, i) 

700 return chain
